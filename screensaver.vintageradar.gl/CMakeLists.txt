cmake_minimum_required(VERSION 3.16)
project(screensaver_vintageradar CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# Vintage Radar (OpenGL): a screensaver add-on for Kodi.
#   cmake -S . -B build -DRADAR_HARNESS=ON
#   cmake -S . -B build -DRADAR_KODI=ON -DKODI_DEVKIT_INCLUDE=<...>
option(RADAR_GLES "Compile the renderer for OpenGL ES 2.0" OFF)
option(RADAR_HARNESS "Build the GLFW test window (needs GLFW)" ON)
option(RADAR_FETCH_GLFW "Download GLFW instead of using an installed one" OFF)
option(RADAR_KODI "Build the Kodi add-on (needs the kodi-addon-dev-kit headers)" OFF)
set(KODI_DEVKIT_INCLUDE "" CACHE PATH "xbmc/addons/kodi-dev-kit/include")

find_package(Threads REQUIRED)

add_library(radar_core STATIC
  src/radar_core.cpp
  src/radar_feed.cpp
)
target_include_directories(radar_core PUBLIC src)
target_link_libraries(radar_core PUBLIC Threads::Threads)
if(NOT MSVC)
  target_compile_options(radar_core PRIVATE -Wall -Wextra)
endif()

enable_testing()
add_executable(test_radar tests/test_radar.cpp)
target_link_libraries(test_radar radar_core)
if(NOT MSVC)
  target_compile_options(test_radar PRIVATE -Wall -Wextra)
endif()
add_test(NAME radar_core COMMAND test_radar ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures)

if(RADAR_HARNESS)
  if(RADAR_FETCH_GLFW)
    include(FetchContent)
    set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(glfw)
    set(GLFW_TARGET glfw)
  else()
    find_package(glfw3 REQUIRED)
    set(GLFW_TARGET glfw)
  endif()
  find_package(OpenGL REQUIRED)

  add_library(radar_render STATIC src/radar_app.cpp src/stb_image_impl.cpp)
  target_link_libraries(radar_render PUBLIC radar_core OpenGL::GL)
  if(RADAR_GLES)
    target_compile_definitions(radar_render PUBLIC RADAR_GLES)
  endif()
  add_executable(radar_glfw src/main_radar_glfw.cpp)
  target_link_libraries(radar_glfw radar_render ${GLFW_TARGET})

  if(APPLE)
    # The same renderer with the 3.2 core profile Kodi runs on macOS.
    add_library(radar_render_core STATIC src/radar_app.cpp src/stb_image_impl.cpp)
    target_link_libraries(radar_render_core PUBLIC radar_core OpenGL::GL)
    target_compile_definitions(radar_render_core PUBLIC RADAR_CORE)
    add_executable(radar_glfw_core src/main_radar_glfw.cpp)
    target_link_libraries(radar_glfw_core radar_render_core ${GLFW_TARGET})
  endif()
endif()

if(RADAR_KODI)
  if(NOT KODI_DEVKIT_INCLUDE)
    find_package(Kodi QUIET)
    if(KODI_INCLUDE_DIR)
      set(KODI_DEVKIT_INCLUDE ${KODI_INCLUDE_DIR}/..)
    else()
      message(FATAL_ERROR "Set -DKODI_DEVKIT_INCLUDE=<xbmc>/xbmc/addons/kodi-dev-kit/include")
    endif()
  endif()
  if(NOT ANDROID)
    find_package(OpenGL REQUIRED)
  endif()

  add_library(screensaver.vintageradar.gl MODULE
    src/kodi_radar.cpp
    src/radar_app.cpp
    src/stb_image_impl.cpp
  )
  target_include_directories(screensaver.vintageradar.gl PRIVATE ${KODI_DEVKIT_INCLUDE})
  target_link_libraries(screensaver.vintageradar.gl PRIVATE radar_core)
  target_compile_definitions(screensaver.vintageradar.gl PRIVATE RADAR_KODI)
  set_target_properties(screensaver.vintageradar.gl PROPERTIES PREFIX "")
  if(NOT MSVC)
    target_compile_options(screensaver.vintageradar.gl PRIVATE -Wall -Wextra)
  endif()
  if(ANDROID)
    target_compile_definitions(screensaver.vintageradar.gl PRIVATE TARGET_ANDROID TARGET_LINUX
                                                                   HAS_GLES=2 RADAR_GLES)
    target_link_libraries(screensaver.vintageradar.gl PRIVATE GLESv2 log)
    set_target_properties(screensaver.vintageradar.gl PROPERTIES SUFFIX ".so")
  elseif(APPLE)
    target_compile_definitions(screensaver.vintageradar.gl PRIVATE TARGET_DARWIN HAS_GL RADAR_CORE
                                                                   GL_SILENCE_DEPRECATION)
    set_target_properties(screensaver.vintageradar.gl PROPERTIES SUFFIX ".dylib")
    target_link_options(screensaver.vintageradar.gl PRIVATE -undefined dynamic_lookup)
    target_link_libraries(screensaver.vintageradar.gl PRIVATE OpenGL::GL)
  elseif(WIN32)
    message(FATAL_ERROR "Kodi on Windows uses ANGLE (GLES); build with the kodi-addons cmake there")
  elseif(RADAR_GLES)
    target_compile_definitions(screensaver.vintageradar.gl PRIVATE TARGET_LINUX HAS_GLES=2 RADAR_GLES)
    target_link_libraries(screensaver.vintageradar.gl PRIVATE GLESv2)
  else()
    target_compile_definitions(screensaver.vintageradar.gl PRIVATE TARGET_LINUX HAS_GL RADAR_CORE)
    target_link_libraries(screensaver.vintageradar.gl PRIVATE OpenGL::GL)
  endif()
endif()
