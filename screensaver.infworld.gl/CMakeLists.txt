cmake_minimum_required(VERSION 3.16)
project(screensaver_infworld CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# Endless World (OpenGL): its own project, separate from screensaver.sandclock.gl.
#   cmake -S . -B build -DSANDCLOCK_HARNESS=ON          # GLFW test window + tests
#   cmake -S . -B build -DSANDCLOCK_KODI=ON -DKODI_DEVKIT_INCLUDE=<...>  # the add-on library
option(SANDCLOCK_GLES "Compile the renderer for OpenGL ES 2.0" OFF)
option(SANDCLOCK_HARNESS "Build the GLFW test window (needs GLFW)" ON)
option(SANDCLOCK_FETCH_GLFW "Download GLFW instead of using an installed one" OFF)
option(SANDCLOCK_KODI "Build the Kodi add-on (needs the kodi-addon-dev-kit headers)" OFF)
set(KODI_DEVKIT_INCLUDE "" CACHE PATH "xbmc/addons/kodi-dev-kit/include, if Kodi's cmake package is not installed")

add_library(infworld_core STATIC src/infworld.cpp)
target_include_directories(infworld_core PUBLIC src)
target_compile_options(infworld_core PRIVATE -Wall -Wextra)

enable_testing()
add_executable(test_infworld tests/test_infworld.cpp)
target_link_libraries(test_infworld infworld_core)
add_test(NAME infworld COMMAND test_infworld)

if(SANDCLOCK_HARNESS)
  if(SANDCLOCK_FETCH_GLFW)
    include(FetchContent)
    set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(glfw)
    set(GLFW_TARGET glfw)
  else()
    find_package(glfw3 REQUIRED)
    set(GLFW_TARGET glfw)
  endif()
  find_package(OpenGL REQUIRED)

  add_library(infworld_render STATIC src/infworld_app.cpp src/stb_image_impl.cpp)
  target_link_libraries(infworld_render PUBLIC infworld_core OpenGL::GL)
  if(SANDCLOCK_GLES)
    target_compile_definitions(infworld_render PUBLIC SANDCLOCK_GLES)
  endif()
  add_executable(infworld_glfw src/main_infworld_glfw.cpp)
  target_link_libraries(infworld_glfw infworld_render ${GLFW_TARGET})

  if(APPLE)
    # The same renderer with the 3.2 core profile Kodi runs on macOS.
    add_library(infworld_render_core STATIC src/infworld_app.cpp src/stb_image_impl.cpp)
    target_link_libraries(infworld_render_core PUBLIC infworld_core OpenGL::GL)
    target_compile_definitions(infworld_render_core PUBLIC SANDCLOCK_CORE)
    add_executable(infworld_glfw_core src/main_infworld_glfw.cpp)
    target_link_libraries(infworld_glfw_core infworld_render_core ${GLFW_TARGET})
  endif()
endif()

# ---- Kodi add-on -----------------------------------------------------------------
if(SANDCLOCK_KODI)
  if(NOT KODI_DEVKIT_INCLUDE)
    find_package(Kodi QUIET)
    if(KODI_INCLUDE_DIR)
      set(KODI_DEVKIT_INCLUDE ${KODI_INCLUDE_DIR}/..)
    else()
      message(FATAL_ERROR "Set -DKODI_DEVKIT_INCLUDE=<xbmc>/xbmc/addons/kodi-dev-kit/include")
    endif()
  endif()
  if(NOT ANDROID)
    find_package(OpenGL REQUIRED)
  endif()

  add_library(screensaver.infworld.gl MODULE src/kodi_infworld.cpp src/infworld_app.cpp
                                              src/stb_image_impl.cpp)
  target_include_directories(screensaver.infworld.gl PRIVATE ${KODI_DEVKIT_INCLUDE})
  target_link_libraries(screensaver.infworld.gl PRIVATE infworld_core)
  target_compile_definitions(screensaver.infworld.gl PRIVATE SANDCLOCK_KODI)
  set_target_properties(screensaver.infworld.gl PROPERTIES PREFIX "" POSITION_INDEPENDENT_CODE ON)
  target_compile_options(screensaver.infworld.gl PRIVATE -Wall -Wextra)
  if(ANDROID)
    # cmake -DCMAKE_TOOLCHAIN_FILE=$ANDROID_NDK_HOME/build/cmake/android.toolchain.cmake
    #       -DANDROID_ABI=arm64-v8a|armeabi-v7a -DANDROID_PLATFORM=android-21 -DANDROID_STL=c++_static
    target_compile_definitions(screensaver.infworld.gl PRIVATE TARGET_ANDROID TARGET_LINUX
                                                             HAS_GLES=2 SANDCLOCK_GLES)
    target_link_libraries(screensaver.infworld.gl PRIVATE GLESv2 log)
    set_target_properties(screensaver.infworld.gl PROPERTIES SUFFIX ".so")
  elseif(APPLE)
    target_compile_definitions(screensaver.infworld.gl PRIVATE TARGET_DARWIN HAS_GL
                                                             SANDCLOCK_CORE GL_SILENCE_DEPRECATION)
    set_target_properties(screensaver.infworld.gl PROPERTIES SUFFIX ".dylib")
    target_link_options(screensaver.infworld.gl PRIVATE -undefined dynamic_lookup)
    target_link_libraries(screensaver.infworld.gl PRIVATE OpenGL::GL)
  elseif(WIN32)
    message(FATAL_ERROR "Kodi on Windows uses ANGLE (GLES); build with the kodi-addons cmake there")
  else()
    target_compile_definitions(screensaver.infworld.gl PRIVATE TARGET_LINUX)
    if(SANDCLOCK_GLES)
      target_compile_definitions(screensaver.infworld.gl PRIVATE HAS_GLES=2)
    else()
      target_compile_definitions(screensaver.infworld.gl PRIVATE HAS_GL)
      target_link_libraries(screensaver.infworld.gl PRIVATE OpenGL::GL)
    endif()
  endif()
endif()
